cmake_minimum_required(VERSION 3.15)
project(AuraGSM LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This release supports Windows only")
endif()
add_executable(AuraGSM WIN32 src/main.cpp src/NativeWindow.cpp)
target_include_directories(AuraGSM PRIVATE third_party/webview2)
target_link_libraries(AuraGSM PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/webview2/WebView2LoaderStatic.lib" advapi32 ole32 oleaut32 user32 gdi32 shell32 shlwapi version)
add_executable(GSMFixture tests/fake_ds.cpp)
foreach(target AuraGSM GSMFixture)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE third_party)
  target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601)
  target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
  target_link_libraries(${target} PRIVATE ws2_32 bcrypt psapi)
endforeach()
add_custom_command(TARGET AuraGSM POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/web" "$<TARGET_FILE_DIR:AuraGSM>/web")
